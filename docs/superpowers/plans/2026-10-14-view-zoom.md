# View Zoom Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Horizontal zoom (wider keys, fewer on screen) and vertical zoom (seconds of music visible) on the Play and Free play screens, driven by pinch/drag gestures, floating buttons and the mouse wheel, remembered per screen, with Play following the notes while zoomed.

**Architecture:** A pure `Viewport` (render/) owns the zoom state in white-key units and lays the keyboard out at the zoomed width; the renderer draws with a `scrollX` offset and a `seconds` window. A pure `followTarget` decides where Play's view should scroll. `ViewGestures` (ui/) turns pointer and wheel events over the notes area into zoom/pan calls; the controllers wire it all together and report button state to a shared `ViewControls.svelte`.

**Tech Stack:** TypeScript, Svelte 5 (runes), Canvas 2D, Vitest (node environment), svelte-check.

**Spec:** `docs/superpowers/specs/2026-10-14-view-zoom-design.md`

## Global Constraints

- Seconds visible: `MIN_SECONDS = 1.5`, `MAX_SECONDS = 8`, `DEFAULT_SECONDS = 3`.
- White keys visible: at least `MIN_WHITE_KEYS = 8` (or the whole range if it has fewer), at most the whole range; saved as `whiteKeys: number | null` (`null` = whole range), valid saved values 8–52.
- One zoom-button press multiplies or divides by `ZOOM_STEP = 1.25`.
- Gestures only for pointers that start above `keyboardTop`; the keyboard stays for playing.
- Drag threshold 6 px; pinch axis span floor 24 px; wheel factor `Math.exp(delta * 0.0025)`; wheel line mode × 16 px; wheel commit 300 ms after the last wheel event.
- Follow: margin of 1 white key each side; glide `scrollX += (target − scrollX) · (1 − e^(−dt/150))`; first follow snaps.
- Settings saved on gesture end, button press, and wheel commit — never on every pointer move.
- The zoom adds no per-frame object allocations to the draw loop; the keyboard layout is rebuilt only when the zoom, range or size changes.
- Code comments, identifiers and UI copy in English; Spanish translations in `src/i18n/es.ts`. Match the surrounding style (short doc comments, no semicolon-less code, 2-space indent, single quotes).
- Run tests with `npx vitest run <file>`; the full suite with `npm test`; types with `npm run check`. Baseline: 195 tests passing, `check` clean.

- [ ] **Step 4: Write the Viewport**

Create `src/render/viewport.ts`:

```ts
import { isBlack, layoutKeyboard, type KeyboardLayout } from './keyboard-layout';

/** Song seconds visible above the keyboard by default, and the vertical zoom limits. */
export const DEFAULT_SECONDS = 3;
export const MIN_SECONDS = 1.5;
export const MAX_SECONDS = 8;
/** Fewest white keys on screen (unless the range itself has fewer). */
export const MIN_WHITE_KEYS = 8;
/** White keys on an 88-key piano. */
export const ALL_WHITE_KEYS = 52;
/** One press of a zoom button multiplies or divides by this. */
export const ZOOM_STEP = 1.25;

const EPSILON = 1e-6;

export interface ViewZoom {
  /** Most white keys on screen; null = the whole keyboard range. */
  whiteKeys: number | null;
  /** Song seconds visible above the keyboard. */
  seconds: number;
}

const clamp = (value: number, min: number, max: number) => Math.min(max, Math.max(min, value));

function whiteKeysBetween(low: number, high: number): number {
  let count = 0;
  for (let p = low; p <= high; p++) if (!isBlack(p)) count++;
  return count;
}

/**
 * What part of the keyboard and of the song is on screen. The horizontal state is kept
 * in white keys (left edge and how many are visible) so it survives resizes; `layout`
 * lays the whole range out at the zoomed key width, in content pixels.
 */
export class Viewport {
  private low: number;
  private high: number;
  private whiteCount: number;
  /** White keys on screen (fractional). */
  private visible: number;
  /** Left edge of the view, in white keys from `low`. */
  private left = 0;
  private width = 0;
  /** The saved `whiteKeys`; replaced only when what is visible changes. */
  private stored: number | null;
  private secondsVisible: number;
  private currentLayout: KeyboardLayout;

  constructor(low: number, high: number, zoom: ViewZoom) {
    this.low = low;
    this.high = high;
    this.whiteCount = whiteKeysBetween(low, high);
    this.stored = zoom.whiteKeys;
    this.visible = this.effectiveVisible();
    this.secondsVisible = clamp(zoom.seconds, MIN_SECONDS, MAX_SECONDS);
    this.currentLayout = this.buildLayout();
  }

  /** The whole range at the zoomed key width; key x in content pixels. */
  get layout(): KeyboardLayout {
    return this.currentLayout;
  }

  get viewWidth(): number {
    return this.width;
  }

  get seconds(): number {
    return this.secondsVisible;
  }

  get visibleWhites(): number {
    return this.visible;
  }

  /** True when part of the range is off screen. */
  get zoomed(): boolean {
    return this.visible < this.whiteCount - EPSILON;
  }

  /** Content pixels scrolled off the left edge. */
  get scrollX(): number {
    return this.left * this.whiteWidth;
  }

  get maxScroll(): number {
    return (this.whiteCount - this.visible) * this.whiteWidth;
  }

  private get whiteWidth(): number {
    return this.visible > 0 ? this.width / this.visible : 0;
  }

  private get minVisible(): number {
    return Math.min(MIN_WHITE_KEYS, this.whiteCount);
  }

  resize(width: number): void {
    if (width === this.width) return;
    this.width = width;
    this.currentLayout = this.buildLayout();
  }

  /** Changes the key range, keeping the saved zoom and the key in the middle. */
  setRange(low: number, high: number): void {
    const center = this.centerPitch();
    this.low = low;
    this.high = high;
    this.whiteCount = whiteKeysBetween(low, high);
    this.visible = this.effectiveVisible();
    this.centerOn(center);
    this.currentLayout = this.buildLayout();
  }

  /**
   * Shows `factor` times as many white keys, keeping the content under `anchorX` (view
   * pixels) in place. Returns whether anything changed.
   */
  zoomKeys(factor: number, anchorX: number): boolean {
    const next = clamp(this.visible * factor, this.minVisible, this.whiteCount);
    if (Math.abs(next - this.visible) < EPSILON) return false;
    const fraction = this.width > 0 ? clamp(anchorX / this.width, 0, 1) : 0.5;
    const anchor = this.left + fraction * this.visible;
    this.visible = next;
    this.stored = next >= this.whiteCount - EPSILON ? null : next;
    this.setLeft(anchor - fraction * next);
    this.currentLayout = this.buildLayout();
    return true;
  }

  /** Shows `factor` times as many seconds. Returns whether anything changed. */
  zoomTime(factor: number): boolean {
    const next = clamp(this.secondsVisible * factor, MIN_SECONDS, MAX_SECONDS);
    if (Math.abs(next - this.secondsVisible) < EPSILON) return false;
    this.secondsVisible = next;
    return true;
  }

  /** Moves the content by dx view pixels, as a drag does. */
  panBy(dx: number): void {
    if (this.whiteWidth > 0) this.setLeft(this.left - dx / this.whiteWidth);
  }

  /** Scrolls so the content pixel x is at the left edge. */
  scrollTo(x: number): void {
    if (this.whiteWidth > 0) this.setLeft(x / this.whiteWidth);
  }

  /** Puts `pitch` in the middle of the view, as far as the range allows. Works before the first resize. */
  centerOn(pitch: number): void {
    const p = clamp(Math.round(pitch), this.low, this.high);
    const whitesBefore = whiteKeysBetween(this.low, p - 1);
    this.setLeft(whitesBefore + (isBlack(p) ? 0 : 0.5) - this.visible / 2);
  }

  /** The white key in the middle of the view. */
  centerPitch(): number {
    const target = Math.min(this.whiteCount - 1, Math.floor(this.left + this.visible / 2));
    let index = 0;
    for (let p = this.low; p <= this.high; p++) {
      if (isBlack(p)) continue;
      if (index === target) return p;
      index++;
    }
    return this.high;
  }

  /** Back to the whole range and the default seconds. */
  fit(): void {
    this.stored = null;
    this.visible = this.whiteCount;
    this.left = 0;
    this.secondsVisible = DEFAULT_SECONDS;
    this.currentLayout = this.buildLayout();
  }

  /** What to save. */
  zoom(): ViewZoom {
    return { whiteKeys: this.stored, seconds: this.secondsVisible };
  }

  /** 1 = zoom in (fewer, wider keys), -1 = zoom out. */
  canZoomKeys(direction: 1 | -1): boolean {
    return direction === 1 ? this.visible > this.minVisible + EPSILON : this.zoomed;
  }

  /** 1 = zoom in (fewer seconds, notes further apart), -1 = zoom out. */
  canZoomTime(direction: 1 | -1): boolean {
    return direction === 1
      ? this.secondsVisible > MIN_SECONDS + EPSILON
      : this.secondsVisible < MAX_SECONDS - EPSILON;
  }

  private effectiveVisible(): number {
    return this.stored === null ? this.whiteCount : clamp(this.stored, this.minVisible, this.whiteCount);
  }

  private setLeft(left: number): void {
    this.left = clamp(left, 0, this.whiteCount - this.visible);
  }

  private buildLayout(): KeyboardLayout {
    return layoutKeyboard(this.low, this.high, this.whiteWidth * this.whiteCount);
  }
}
```

```ts
import type { PlayableNote, PlayRole } from '../song/types';
import { lowerBound } from '../util/bisect';
import type { Viewport } from './viewport';

/**
 * Where the view should scroll so the next notes of `roles` are on screen, or null to
 * stay. Takes the notes on screen by start time while they fit (with a white key of margin
 * each side) and moves as little as possible to show them. Notes starting together are
 * sorted by pitch, so a chord wider than the view shows its lowest notes.
 */
export function followTarget(
  notes: readonly PlayableNote[],
  maxDuration: number,
  roles: ReadonlySet<PlayRole>,
  songTime: number,
  view: Viewport,
): number | null {
  const width = view.viewWidth;
  if (width <= 0) return null;
  const layout = view.layout;
  const margin = layout.whiteWidth;
  const from = lowerBound(notes, songTime - maxDuration);
  const to = lowerBound(notes, songTime + view.seconds + 0.001);
  let left = Infinity;
  let right = -Infinity;
  for (let i = from; i < to; i++) {
    const n = notes[i];
    if (n.end <= songTime || !roles.has(n.role)) continue;
    const key = layout.byPitch.get(n.pitch);
    if (!key) continue;
    const l = Math.min(left, key.x);
    const r = Math.max(right, key.x + key.width);
    if (left !== Infinity && r - l + 2 * margin > width) break;
    left = l;
    right = r;
  }
  if (left === Infinity) return null;

  const scrollX = view.scrollX;
  let target: number;
  if (right - left + 2 * margin > width) target = (left + right - width) / 2;
  else if (left < scrollX + margin) target = left - margin;
  else if (right > scrollX + width - margin) target = right + margin - width;
  else return null;
  target = Math.min(view.maxScroll, Math.max(0, target));
  return Math.abs(target - scrollX) < 0.5 ? null : target;
}
```

```ts
import { geometry } from '../render/renderer';
import type { KeyboardSurface } from './keyboard-input';

/** What the gestures ask of the view. Factors multiply the white keys or seconds shown (< 1 zooms in). */
export interface ViewGestureHandler {
  zoomKeys(factor: number, anchorX: number): void;
  zoomTime(factor: number): void;
  /** Moves the content by dx canvas pixels, as a drag does. */
  pan(dx: number): void;
  /** A gesture ended, or the wheel went quiet: save the zoom. */
  commit(): void;
}

/** Sideways movement before one finger starts panning. */
const DRAG_THRESHOLD = 6;
/** A pinch is never measured narrower than this, so the factor cannot explode. */
const MIN_PINCH_SPAN = 24;
const WHEEL_ZOOM = 0.0025;
const WHEEL_LINE_PX = 16;
const WHEEL_COMMIT_MS = 300;

interface Point {
  x: number;
  y: number;
}

/**
 * Pinch, drag and wheel over the falling notes as zoom and pan. Pointers that start on
 * the keyboard are left to KeyboardInput.
 */
export class ViewGestures {
  /** Pointers that started over the notes (at most two), at their last used position. */
  private points = new Map<number, Point>();
  private pinchAxis: 'x' | 'y' | null = null;
  private dragging = false;
  private wheelTimer: ReturnType<typeof setTimeout> | null = null;

  constructor(
    private readonly canvas: HTMLCanvasElement,
    private readonly surface: () => KeyboardSurface | null,
    private readonly handler: ViewGestureHandler,
  ) {}

  /** Starts listening; returns the cleanup. */
  attach(): () => void {
    const cleanups: Array<() => void> = [];
    const listen = (type: string, handler: (e: never) => void, options?: AddEventListenerOptions) => {
      this.canvas.addEventListener(type, handler as EventListener, options);
      cleanups.push(() => this.canvas.removeEventListener(type, handler as EventListener, options));
    };
    listen('pointerdown', this.onPointerDown);
    listen('pointermove', this.onPointerMove);
    listen('pointerup', this.onPointerUp);
    listen('pointercancel', this.onPointerUp);
    listen('wheel', this.onWheel, { passive: false });
    return () => {
      for (const cleanup of cleanups) cleanup();
      this.points.clear();
      if (this.wheelTimer !== null) {
        clearTimeout(this.wheelTimer);
        this.wheelTimer = null;
        this.handler.commit();
      }
    };
  }

  private local(e: { clientX: number; clientY: number }): Point {
    const rect = this.canvas.getBoundingClientRect();
    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
  }

  private onPointerDown = (e: PointerEvent): void => {
    const surface = this.surface();
    if (!surface || this.points.size >= 2) return;
    const p = this.local(e);
    const { keyboardTop } = geometry(surface.cssWidth, surface.cssHeight, surface.layout.whiteWidth);
    if (p.y >= keyboardTop) return;
    e.preventDefault();
    this.canvas.setPointerCapture?.(e.pointerId);
    this.points.set(e.pointerId, p);
    this.dragging = false;
    if (this.points.size === 2) {
      const [a, b] = [...this.points.values()];
      this.pinchAxis = Math.abs(a.x - b.x) >= Math.abs(a.y - b.y) ? 'x' : 'y';
    } else {
      this.pinchAxis = null;
    }
  };

  private onPointerMove = (e: PointerEvent): void => {
    const prev = this.points.get(e.pointerId);
    if (!prev) return;
    const p = this.local(e);
    if (this.pinchAxis) {
      let other: Point = prev;
      for (const [id, point] of this.points) if (id !== e.pointerId) other = point;
      const axis = this.pinchAxis;
      const before = Math.max(MIN_PINCH_SPAN, Math.abs(prev[axis] - other[axis]));
      const after = Math.max(MIN_PINCH_SPAN, Math.abs(p[axis] - other[axis]));
      this.points.set(e.pointerId, p);
      if (before === after) return;
      if (axis === 'x') this.handler.zoomKeys(before / after, (p.x + other.x) / 2);
      else this.handler.zoomTime(before / after);
      return;
    }
    // Below the threshold the start point is kept, so no movement is lost once dragging starts.
    if (!this.dragging && Math.abs(p.x - prev.x) < DRAG_THRESHOLD) return;
    this.dragging = true;
    this.points.set(e.pointerId, p);
    if (p.x !== prev.x) this.handler.pan(p.x - prev.x);
  };

  private onPointerUp = (e: PointerEvent): void => {
    if (!this.points.delete(e.pointerId)) return;
    // The finger left after a pinch carries on as a drag from where it is.
    this.dragging = this.points.size === 1 && this.pinchAxis !== null;
    this.pinchAxis = null;
    this.handler.commit();
  };

  private onWheel = (e: WheelEvent): void => {
    if (!this.surface()) return;
    e.preventDefault();
    const scale = e.deltaMode === 1 ? WHEEL_LINE_PX : 1;
    const dx = e.deltaX * scale;
    const dy = e.deltaY * scale;
    if (e.ctrlKey) this.handler.zoomKeys(Math.exp(dy * WHEEL_ZOOM), this.local(e).x);
    else if (e.shiftKey) this.handler.pan(-(dx !== 0 ? dx : dy));
    else if (Math.abs(dx) > Math.abs(dy)) this.handler.pan(-dx);
    else if (dy !== 0) this.handler.zoomTime(Math.exp(dy * WHEEL_ZOOM));
    if (this.wheelTimer !== null) clearTimeout(this.wheelTimer);
    this.wheelTimer = setTimeout(() => {
      this.wheelTimer = null;
      this.handler.commit();
    }, WHEEL_COMMIT_MS);
  };
}
```

```ts
import { DEFAULT_SECONDS, type Viewport } from '../render/viewport';

/** What the zoom buttons can do right now. */
export interface ViewState {
  canZoomKeysIn: boolean;
  canZoomKeysOut: boolean;
  canZoomTimeIn: boolean;
  canZoomTimeOut: boolean;
  /** Something differs from the whole keyboard at the default seconds. */
  canFit: boolean;
  /** Play only: zoomed in and not following the notes. */
  showFollow: boolean;
}

export const NO_VIEW_STATE: ViewState = {
  canZoomKeysIn: false,
  canZoomKeysOut: false,
  canZoomTimeIn: false,
  canZoomTimeOut: false,
  canFit: false,
  showFollow: false,
};

export function viewState(view: Viewport, showFollow: boolean): ViewState {
  return {
    canZoomKeysIn: view.canZoomKeys(1),
    canZoomKeysOut: view.canZoomKeys(-1),
    canZoomTimeIn: view.canZoomTime(1),
    canZoomTimeOut: view.canZoomTime(-1),
    canFit: view.zoomed || Math.abs(view.seconds - DEFAULT_SECONDS) > 1e-6,
    showFollow,
  };
}

function sameViewState(a: ViewState, b: ViewState): boolean {
  return (Object.keys(a) as (keyof ViewState)[]).every((key) => a[key] === b[key]);
}

/** Sends the buttons' state to the screen, only when it changes. */
export class ViewReporter {
  private last: ViewState | null = null;

  constructor(private readonly onChange: (state: ViewState) => void) {}

  report(view: Viewport, showFollow = false): void {
    const next = viewState(view, showFollow);
    if (this.last && sameViewState(this.last, next)) return;
    this.last = next;
    this.onChange(next);
  }
}

/** How a controller gets its saved zoom, reports the buttons' state and saves the zoom. */
export interface ViewOptions<Z> {
  zoom: Z;
  onChange: (state: ViewState) => void;
  onCommit: (zoom: Z) => void;
}
```
